\begin{multicols}{2}
    \begin{tcolorbox}[title=Características fuente]
        \begin{tabular}{rl}
            \textbf{Alfabeto}:     & $X = \left\{ x_1, x_2, \ldots x_S \right\}$ \\
            \textbf{Probabilidad}: & $p_i : 0 \leq p_i \leq 1, \sum_{i=1}^{S}{p_i} = 1$ \\
            \textbf{Información}:  & $I_i = \log_2{\frac{1}{p_i}} = -\log_2{p_i}$ \\
                        & $I_i \geq 0$ para cualquier $0 \leq p_i \leq 1$ \\
                        & $I_i \to 0$ cuando $p_i \to 1$ \\
            \textbf{Entropía}:     & $H(X) = \sum_{i=1}^{S}{p_i \cdot I_i} = \sum_{i=1}^{S}{p_i \cdot \log_2{\frac{1}{p_i}}}$ \\
                        & $0 \leq H(X) \leq \log_2{S}$ \\
        \textbf{Longitud media}: & $\overline{L} = \sum_{i=1}^{S}{p_i \cdot L_i}$ \\
                        & en codificación longitud fija $\overline{L} = \lceil\log_2{S}\rceil$ \\
                        & $H(X) \leq \overline{L} < H(X) + \epsilon$ \\
            \textbf{Eficiencia}:   & $\eta = \frac{H(X)}{\overline{L}}$ \\
            \textbf{Kraft}: & código unívocamente descifrable si: \\
                            & $K = \sum\limits_{i=1}^{S}{2^{-L_i}} \leq 1 \implies$ \\
            \textbf{Tasa compresión}: & $r = \frac{\lceil \log_2{S} \rceil}{\overline{L}}$
        \end{tabular}
    \end{tcolorbox}
    \begin{tcolorbox}[title=Entropía binaria]

\begin{tcolorbox}[title=Enfoque probabilístico]
    \begin{tabularx}{\linewidth}{X X}
        \begin{tabular}{rl}

            $p(x_i)$: & probabilidad fuente envíe símbolo $x_i$ \\

        \end{tabular}
        &
        \begin{tabular}{rl}
            \multicolumn{2}{l}{\bfseries Probabilidades condicionadas} \\
            $p(y_j \vert x_i)$: & probabilidad transición hacia delante \\
            $p(x_i \vert y_j)$: & probabilidad transición hacia atrás
        \end{tabular}
        \\\vspace{1pt}
        \begin{tabular}{rl}
            \multicolumn{2}{l}{\bfseries Aplicación teoremas básicos} \\
            Bayes: & $p(x_i,y_j) = p(y_j \vert x_i) \cdot p(x_i) = p(x_i \vert y_j) \cdot p(y_j)$ \\
            Prob.\ total: & $p(y_j) = \sum_{i=1}^{S}{p(y_j \vert x_i) \cdot p(x_i)}$ \\
                          & $p(x_i) = \sum_{i=1}^{N}{p(x_i \vert y_j) \cdot p(y_j)}$
        \end{tabular}
    \end{tabularx}
\end{tcolorbox}

\begin{tcolorbox}[title=Canal discreto]
    \textbf{Información mútua} \\
    Cantidad de información que ha transmitido el canal cuando la fuente envía el símbolo $x_i$ y la fuente recibe el símbolo $y_j$.
    \[
        I(x_i;y_j) = \log_2{\frac{1}{p(x_i)}} - \log_2{\frac{1}{p(x_i \vert y_j)}} = \log_2{\frac{p(x_i \vert y_j)}{p(x_i)}} = \log_2{\frac{p(y_j \vert x_i)}{p(y_j)}}
    \]

    \textbf{Información mútua media} \\
    \[
        I(X;Y) = \sum_{i=1}^{S}\sum_{j=1}^{N}{p(x_i,y_j) \cdot I(x_i;y_j)}
    \]

    \textbf{Equivocación} \\
    \[
        H(X \vert Y) = \sum\limits_{i=1}^{S}\sum\limits_{j=1}^{N}{p(x_i,y_j)\log_2{\frac{1}{p(x_i \vert y_j)}}}
    \]

    Al promediar tenemos:
    \[
        \left.
        \begin{array}{rl}
            I(x_i;y_j) \to & I(X;Y) \\
            \log_2{\frac{1}{p(x_i)}} \to & H(X) \\
            \log_2{\frac{1}{p(x_i \vert y_j)}} \to & H(X \vert Y)
        \end{array}
        \right\}
        I(X;Y) = H(X) - H(X \vert Y)
    \]

    \textbf{Capacidad} \\
    Máxima cantidad de bits de información por símbolo
    \[
        C = \max\limits_{p(x_i), p(y_j \vert x_i)}{I(X;Y)}
    \]
\end{tcolorbox}

\begin{tcolorbox}[title=Señal]
    Una señal $s(t)$ es cualquier variación temporal de una magnitud eléctrica, electromagnética, óptica, acústica o química en función del mensaje que se quiere enviar.
    El envío del mensaje tiene lugar con la propagación de la señal a través de un \textbf{medio de transmisión}.
    Las señales pueden ser \textbf{analógicas}/\textbf{digitales} y \textbf{periódicas}/\textbf{no periódicas}; se caracterizan por su \textbf{espectro} y \textbf{potencia}.
    \[
        s(t) = A \sin(2\pi f t + \phi)
    \]
    \begin{itemize}
        \item \textbf{Amplitud $A$}: valor de pico (máximo desplazamiento en positivo)
        \item \textbf{Período $T$}: tiempo entre picos (e.q. tiempo de duración de una oscilación).
        \item \textbf{Frecuencia $f$}: numero de oscilaciones por unidad de tiempo ($f = \frac{1}{T}$).
        \item \textbf{Fase $\phi$}: posición relativa con respecto al origen de tiempos. Convenios: grados ($[0, 360)$ o $[-180, 180)$) o radianes ($[0, 2\pi)$ o $[-\pi, \pi)$).
    \end{itemize}
    \textbf{Potencia}
    \[
        p(t) = v(t)i(t) = v(t)\frac{v(t)}{R} = \frac{v^2(t)}{R}
    \]
    En general cualquier tipo de señal:
    \[
        p(t) = ks^2(t)
    \]
    Para una señal de duración finita $\tau$ o de un tramo de duración $\tau$ de una señal de duración infinita:
    \[
        P = \frac{1}{\tau}\int_\tau{p(t)dt} = \frac{1}{\tau}\int_\tau{ks^2(t)dt}
    \]
\end{tcolorbox}

\begin{tcolorbox}[title=Decibelio]
    Mide la relación entre dos potencias (métrica relativa):
    \[
        \frac{P_2}{P_1}(\mathrm{dB}) = 10\log_{10}{\frac{P_2}{P_1}}
    \]
    \begin{multicols}{2}
        Derivaciones absolutas del decibelio:
        \[
            \begin{aligned}
                P(\mathrm{dBm}) = 10\log_{10}{\frac{P(\mathrm{mW})}{1\mathrm{mW}}} \Longleftrightarrow & P(\mathrm{mW}) = 10^{\frac{P(\mathrm{dBm})}{10}} \\
                P(\mathrm{dBW}) = 10\log_{10}{\frac{P(\mathrm{W})}{1\mathrm{W}}} \Longleftrightarrow & P(\mathrm{W}) = 10^{\frac{P(\mathrm{dB})}{10}}
            \end{aligned}
        \]
        \\
        Relación entre dBm y dBW:
        \[
            \begin{aligned}
                10\log_{10}{P(\mathrm{mW})} = & \hspace{2pt} 10\log_{10}{(1000P(W))} \\
                P(\mathrm{dBm}) = & \hspace{2pt} P(dBw) + 30
            \end{aligned}
        \]
        \\
    \end{multicols}
    \begin{multicols}{2}
        \textbf{Amplificador:}
        \[
            \begin{aligned}
                \text{Factor de amplificación: } \alpha = \frac{P_\mathrm{out}}{P_\mathrm{in}} \\
                \text{Ganancia: } G(\textrm{dB}) = 10\log_{10}{\alpha}
            \end{aligned}
        \]
        \textbf{Medio de transmisión:}
        \[
            \begin{aligned}
                \text{Factor de atenuación: } \beta = \frac{P_\mathrm{in}}{P_\mathrm{out}} > 1 \\
                L(\textrm{dB}) = 10\log_{10}{\beta} > 0
            \end{aligned}
        \]
    \end{multicols}

    Ejemplo: cascada de amplificadores
    \[
        P_r = P_t\frac{1}{\beta_1}\alpha_1\frac{1}{\beta_2}\alpha_2\cdots\frac{1}{\beta_n}\alpha_n \implies 10\log_{10}{P_r} = 10\log_{10}P_t - L_1 + G_1 - L2 + G2 \cdots - L_n + G_n
    \]
\end{tcolorbox}

\begin{tcolorbox}[title=Espectro de un a señal]
    El espectro $S$, la potencia $P$ y ancho de banda $B$ de una señal:
    \[
        P = \int_{f_{\textrm{min}}}^{f_{\textrm{max}}}{S(f)df}, B = f_\mathrm{max} - f_\mathrm{min}
    \]
\end{tcolorbox}
